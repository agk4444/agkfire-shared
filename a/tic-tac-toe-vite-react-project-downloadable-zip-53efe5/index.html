"src/hooks/useTicTacToe.js": `import { useCallback, useEffect, useMemo, useReducer } from 'react'
import { createEmptyBoard, isOver, winnerOf, winningLineOf, nextBestMove, DIFFICULTY } from '../game/engine.js'

const initialState = {
  board: createEmptyBoard(),
  turn: 'X',
  history: [],
  mode: 'ai',
  difficulty: 'hard',
  score: { X: 0, O: 0, draw: 0 },
  thinking: false
}

function reducer(state, action) {
  switch (action.type) {
    case 'MOVE': {
      if (isOver(state.board) || state.board[action.index] !== null) return state
      const board = state.board.slice()
      board[action.index] = state.turn
      const done = isOver(board)
      return {
        ...state,
        board,
        history: state.history.concat([{ board: state.board, turn: state.turn, score: state.score }]),
        turn: state.turn === 'X' ? 'O' : 'X',
        thinking: false
      }
    }
    case 'AI_MOVE': {
      if (isOver(state.board) || state.board[action.index] !== null) return state
      const board = state.board.slice()
      board[action.index] = action.mark
      const done = isOver(board)
      return {
        ...state,
        board,
        turn: done ? state.turn : state.turn === 'X' ? 'O' : 'X'
      }
    }
    case 'THINK':
      return { ...state, thinking: action.value }
    case 'UNDO': {
      if (!state.history.length) return state
      const last = state.history[state.history.length - 1]
      return { ...state, board: last.board, turn: last.turn, score: last.score, history: state.history.slice(0, -1), thinking: false }
    }
    case 'SET_MODE':
      return { ...state, mode: action.mode, history: [], thinking: false }
    case 'SET_DIFFICULTY':
      return { ...state, difficulty: action.difficulty, history: [] }
    case 'SCORE':
      return { ...state, score: action.score }
    case 'RESET':
      return { ...state, board: createEmptyBoard(), turn: 'X', history: [], thinking: false }
    case 'CLEAR_SCORES':
      return { ...state, score: { X: 0, O: 0, draw: 0 } }
    default:
      return state
  }
}

const readScore = () => {
  try {
    const raw = localStorage.getItem('ttt-score')
    if (!raw) return initialState.score
    const p = JSON.parse(raw)
    return {
      X: Number(p.X) || 0,
      O: Number(p.O) || 0,
      draw: Number(p.draw) || 0
    }
  } catch {
    return initialState.score
  }
}

export function useTicTacToe() {
  const [state, dispatch] = useReducer(reducer, initialState, (s) => ({ ...s, score: readScore() }))

  useEffect(() => {
    try {
      localStorage.setItem('ttt-score', JSON.stringify(state.score))
    } catch {
      /* storage unavailable - scores just won't persist */
    }
  }, [state.score])

  const over = isOver(state.board)
  const winner = winnerOf(state.board)

  // Score the finished game, then set up the next round.
  useEffect(() => {
    if (!over) return
    const delta = { ...state.score }
    if (winner === null) delta.draw += 1
    else delta[winner] += 1
    if (delta.X !== state.score.X || delta.O !== state.score.O || delta.draw !== state.score.draw) {
      dispatch({ type: 'SCORE', score: delta })
    }
    const t = setTimeout(() => dispatch({ type: 'RESET' }), 1100)
    return () => clearTimeout(t)
  }, [over, winner, state.score])

  // Let the AI answer on the human's turn.
  useEffect(() => {
    if (state.mode !== 'ai' || over || state.turn !== 'O') return
    dispatch({ type: 'THINK', value: true })
    const move = nextBestMove(state.board, 'O', state.difficulty)
    const t = setTimeout(() => {
      if (move !== null) dispatch({ type: 'AI_MOVE', index: move, mark: 'O' })
    }, 260)
    return () => clearTimeout(t)
  }, [state.mode, state.board, state.turn, state.difficulty, over])

  const play = useCallback((index) => dispatch({ type: 'MOVE', index }), [])
  const undo = useCallback(() => dispatch({ type: 'UNDO' }), [])
  const newRound = useCallback(() => dispatch({ type: 'RESET' }), [])
  const clearScores = useCallback(() => dispatch({ type: 'CLEAR_SCORES' }), [])
  const setMode = useCallback((mode) => dispatch({ type: 'SET_MODE', mode }), [])
  const setDifficulty = useCallback((difficulty) => dispatch({ type: 'SET_DIFFICULTY', difficulty }), [])

  const statusText = useMemo(() => {
    if (!over) return state.mode === 'ai' ? (state.turn === 'X' ? 'Your move (X)' : 'Computer thinking…') : 'Player ' + state.turn
    if (winner === null) return 'Draw'
    if (state.mode === 'ai') return winner === 'X' ? 'You win!' : 'Computer wins'
    return 'Player ' + winner + ' wins'
  }, [over, winner, state.mode, state.turn])

  return {
    board: state.board,
    turn: state.turn,
    mode: state.mode,
    difficulty: state.difficulty,
    score: state.score,
    history: state.history,
    canUndo: state.history.length > 0,
    isThinking: state.thinking,
    winningLine: winningLineOf(state.board),
    lastMove: state.history.length ? state.history[state.history.length - 1] : null,
    statusText,
    statusKind: over ? (winner ? 'win' : 'draw') : 'play',
    play,
    undo,
    newRound,
    clearScores,
    setMode,
    setDifficulty
  }
}

export { DIFFICULTY }

"src/game/engine.js": `// Pure game logic - no React, no DOM. Fully unit tested.

export const EMPTY = null
export const MARKS = ['X', 'O']

export const LINES = [
  [0, 1, 2], [3, 4, 5], [6, 7, 8],
  [0, 3, 6], [1, 4, 7], [2, 5, 8],
  [0, 4, 8], [2, 4, 6]
]

export const DIFFICULTY = {
  easy:   { label: 'Easy',   randomChance: 0.75, depth: 2 },
  medium: { label: 'Medium', randomChance: 0.30, depth: 4 },
  hard:   { label: 'Hard',   randomChance: 0,    depth: 9 }
}

export function createEmptyBoard() {
  return Array(9).fill(EMPTY)
}

export function winnerOf(board) {
  for (const [a, b, c] of LINES) {
    if (board[a] && board[a] === board[b] && board[a] === board[c]) return board[a]
  }
  return null
}

export function winningLineOf(board) {
  for (const [a, b, c] of LINES) {
    if (board[a] && board[a] === board[b] && board[a] === board[c]) return [a, b, c]
  }
  return null
}

export function isFull(board) {
  return board.every((c) => c !== EMPTY)
}

export function isOver(board) {
  return winnerOf(board) !== null || isFull(board)
}

export function emptyIndices(board) {
  const out = []
  board.forEach((v, i) => { if (v === EMPTY) out.push(i) })
  return out
}

/**
 * Negamax with alpha-beta pruning and a transposition table.
 * Returns a score for the side-to-move, memoised per (board, mark, depth).
 */
const memo = new Map()

function key(board, mark, depth) {
  let s = mark
  for (let i = 0; i < 9; i++) s += board[i] === EMPTY ? '_' : board[i]
  return s + '|' + depth
}

function negamax(board, mark, depth, alpha, beta, rand) {
  const winner = winnerOf(board)
  if (winner) return winner === mark ? 10 - (9 - emptyIndices(board).length) : -10 + (9 - emptyIndices(board).length)
  const empties = emptyIndices(board)
  if (!empties.length) return 0
  if (depth <= 0) return 0

  const k = key(board, mark, depth)
  const cached = memo.get(k)
  if (cached !== undefined) return cached

  let best = -Infinity
  const order = rand ? shuffle(empties, rand) : empties
  const other = mark === 'X' ? 'O' : 'X'

  for (const i of order) {
    board[i] = mark
    const score = -negamax(board, other, depth - 1, -beta, -alpha, rand)
    board[i] = EMPTY
    if (score > best) best = score
    if (best > alpha) alpha = best
    if (alpha >= beta) break
  }

  memo.set(k, best)
  return best
}

function shuffle(arr, rand) {
  const a = arr.slice()
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(rand() * (i + 1))
    ;[a[i], a[j]] = [a[j], a[i]]
  }
  return a
}

/**
 * Choose a move for \`mark\`.
 * Random tiers mix in random play by probability; hard is pure minimax (never loses).
 */
export function nextBestMove(board, mark, difficulty = 'hard', rand = Math.random) {
  const empties = emptyIndices(board)
  if (!empties.length) return null

  const cfg = DIFFICULTY[difficulty] || DIFFICULTY.hard
  if (rand() < cfg.randomChance) return empties[Math.floor(rand() * empties.length)]

  const working = board.slice()
  const scored = empties.map((i) => {
    working[i] = mark
    const score = -negamax(working, mark === 'X' ? 'O' : 'X', cfg.depth - 1, -Infinity, Infinity, rand)
    working[i] = EMPTY
    return { i, score }
  })

  const best = Math.max(...scored.map((s) => s.score))
  const ties = scored.filter((s) => s.score === best).map((s) => s.i)
  return ties[Math.floor(rand() * ties.length)]
}

export function clearMemo() {
  memo.clear()
}
`,

    board[move] = 'O'
    expect(winnerOf(board)).not.toBe('X')

## How the AI works

- Easy — 75 percent random moves
- Medium — 30 percent random moves, otherwise depth 4 search
- Hard — full depth 9 search, no randomness. It cannot lose; at best it draws.

Tie-breaks are randomised, so Hard plays differently between games while keeping the same result.

## Notes

The reducer scores a finished round and auto-clears the board after 1.1s. Undo restores the
previous board, turn and score snapshot. Scores persist to localStorage.

/* ---------------- minimal store-only ZIP writer ---------------- */
const CRC_TABLE = (() => {
  const t = new Uint32Array(256)
  for (let n = 0; n < 256; n++) {
    let c = n
    for (let k = 0; k < 8; k++) c = c & 1 ? 0xedb88320 ^ (c >>> 1) : c >>> 1
    t[n] = c >>> 0
  }
  return t
})()

function crc32(bytes) {
  let c = 0xffffffff
  for (let i = 0; i < bytes.length; i++) c = CRC_TABLE[(c ^ bytes[i]) & 0xff] ^ (c >>> 8)
  return (c ^ 0xffffffff) >>> 0
}

function dosDateTime(d) {
  const time = ((d.getHours() & 31) << 11) | ((d.getMinutes() & 63) << 5) | ((d.getSeconds() / 2) & 31)
  const date = (((d.getFullYear() - 1980) & 127) << 9) | (((d.getMonth() + 1) & 15) << 5) | (d.getDate() & 31)
  return { time, date }
}

function makeZip(entries) {
  const enc = new TextEncoder()
  const chunks = []
  const central = []
  let offset = 0
  const { time, date } = dosDateTime(new Date())

  for (const [name, content] of entries) {
    const nameBytes = enc.encode(name)
    const data = enc.encode(content)
    const crc = crc32(data)

    const local = new DataView(new ArrayBuffer(30))
    local.setUint32(0, 0x04034b50, true)
    local.setUint16(4, 20, true)
    local.setUint16(6, 0, true)
    local.setUint16(8, 0, true)
    local.setUint16(10, time, true)
    local.setUint16(12, date, true)
    local.setUint32(14, crc, true)
    local.setUint32(18, data.length, true)
    local.setUint32(22, data.length, true)
    local.setUint16(26, nameBytes.length, true)
    local.setUint16(28, 0, true)

    chunks.push(new Uint8Array(local.buffer), nameBytes, data)

    const cen = new DataView(new ArrayBuffer(46))
    cen.setUint32(0, 0x02014b50, true)
    cen.setUint16(4, 20, true)
    cen.setUint16(6, 20, true)
    cen.setUint16(8, 0, true)
    cen.setUint16(10, 0, true)
    cen.setUint16(12, time, true)
    cen.setUint16(14, date, true)
    cen.setUint32(16, crc, true)
    cen.setUint32(20, data.length, true)
    cen.setUint32(24, data.length, true)
    cen.setUint16(28, nameBytes.length, true)
    cen.setUint32(42, offset, true)
    central.push(new Uint8Array(cen.buffer), nameBytes)

    offset += 30 + nameBytes.length + data.length
  }

  const centralSize = central.reduce((n, c) => n + c.length, 0)
  const end = new DataView(new ArrayBuffer(22))
  end.setUint32(0, 0x06054b50, true)
  end.setUint16(8, entries.length, true)
  end.setUint16(10, entries.length, true)
  end.setUint32(12, centralSize, true)
  end.setUint32(16, offset, true)

  return new Blob([...chunks, ...central, new Uint8Array(end.buffer)], { type: 'application/zip' })
}

document.getElementById('dl').addEventListener('click', () => {
  const msg = document.getElementById('msg')
  try {
    const blob = makeZip(Object.entries(FILES))
    const url = URL.createObjectURL(blob)
    const a = document.createElement('a')
    a.href = url
    a.download = 'tic-tac-toe-react.zip'
    document.body.appendChild(a)
    a.click()
    a.remove()
    setTimeout(() => URL.revokeObjectURL(url), 4000)
    msg.innerHTML = 'Downloaded ' + names.length + ' files (' + Math.round(blob.size / 1024) + ' KB). Extract, then run npm install && npm run dev.'
  } catch (e) {
    msg.textContent = 'ZIP failed: ' + e.message
  }
})
